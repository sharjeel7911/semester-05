void runCaesar() {
	string text;
	int key;

	cout << "\n--- Caesar Cipher ---" << endl;

	cout << "Enter text: ";
	getline(cin, text);

	cout << "Enter key (integer): ";
	cin >> key;
	cin.ignore();

	string encrypted = caesarEncrypt(text, key);
	string decrypted = caesarDecrypt(encrypted, key);

	cout << "Encrypted: " << encrypted << endl;
	cout << "Decrypted: " << decrypted << endl;
}
string caesarEncrypt(string text, int key) {
	for (char& c : text) {
		if (isalpha(c)) {
			char base = isupper(c) ? 'A' : 'a';
			c = (c - base + key) % 26 + base;
		}
	}
	return text;
}
string caesarDecrypt(string text, int key) {
	for (char& c : text) {
		if (isalpha(c)) {
			char base = isupper(c) ? 'A' : 'a';
			c = (c - base - key + 26) % 26 + base;
		}
	}
	return text;
}

string vigenereEncrypt(string text, string key) {
	int j = 0;
	for (char& c : text) {
		if (isalpha(c)) {
			char base = isupper(c) ? 'A' : 'a';
			int shift = tolower(key[j]) - 'a';

			c = ((c - base) + shift) % 26 + base;
			j = (j + 1) % key.length();
		}
	}

}

	getline(cin, text);

	cout << "Enter key (string): ";
	cin >> key;
	cin.ignore();

	string encrypted = autoKeyEncrypt(text, key);
	string decrypted = autoKeyDecrypt(encrypted, key);

	cout << "Encrypted: " << encrypted << endl;
	cout << "Decrypted: " << decrypted << endl;
}
string autoKeyEncrypt(string text, string key) {
	int j = 0;
	string keyStream = key;
	for (char& c : text) {
		if (isalpha(c)) {
			char original = c;
			char base = isupper(c) ? 'A' : 'a';
			int shift = tolower(keyStream[j]) - 'a';

			c = ((c - base) + shift) % 26 + base;
			keyStream += tolower(original);
			j++;
		}
	}
	return text;
}
string autoKeyDecrypt(string text, string key) {
	int j = 0;
	string keyStream = key;
	for (char& c : text) {
		if (isalpha(c)) {
			char base = isupper(c) ? 'A' : 'a';
			int shift = tolower(keyStream[j]) - 'a';

			c = ((c - base) - shift + 26) % 26 + base;
			keyStream += tolower(c);
			j++;
		}
	}
	return text;
}

void printFrequencyTable(const string text) {
	unordered_map<char, int> frequency;

	for (char c : text) {
		if (isalpha(c)) {
			c = tolower(c);
			frequency[c]++;
		}
	}
	cout << "Character Frequency Table:" << endl;
	cout << "--------------------------" << endl;
	for (const auto& pair : frequency) {
		cout << pair.first << ": " << pair.second << endl;
	}
}

void caesarBruteForce(string text) {
    for (int key = 0; key < 26; key++) {
        string result = text;

        for (char& c : result) {
            if (isalpha(c)) {
                char base = isupper(c) ? 'A' : 'a';
                c = (c - base - key + 26) % 26 + base;
            }
        }

        cout << "Key " << key << ": " << result << endl;
    }
}

#include <cctype>

#include <vector>

using namespace std;

// Generate 25-char key matrix (combining I and J)
string prepareKey(string key) {
	string cleaned = "";
	for (char c : key) {
		if (isalpha(c)) {
			c = toupper(c);
			if (c == 'J') {
				c = 'I';
			}
			if (cleaned.find(c) == string::npos) {
				cleaned += c;
			}
		}
	}

	for (char i = 'A'; i <= 'Z'; i++) {
		if (i == 'J') continue;
		if (cleaned.find(i) == string::npos) {
			cleaned += i;
		}
	}
	return cleaned;
}

// Prepare plaintext (handle double letters & odd lengths)
string prepareText(string text) {
	string clean = "";
	for (char c : text) {
		if (isalpha(c)) {
			char u = toupper(c);
			if (u == 'J') u = 'I';
			clean += u;
		}
	}
	string prepared = "";
	for (size_t i = 0; i < clean.length(); i++) {
		prepared += clean[i];
		if (i + 1 < clean.length()) {
			if (clean[i] == clean[i + 1]) {
				prepared += 'X';
			} else {
				prepared += clean[i + 1];
				i++;
			}
		}
	}
	if (prepared.length() % 2 != 0) prepared += 'X';
	return prepared;
}
vector<pair<char, char>> prepareText(const string& text, bool encrypt) {
	string clean = "";
	for (char c : text) {
		if (isalpha(c)) {
			char u = toupper(c);
			clean += (u == 'J') ? 'I' : u;
		}
	}

	vector<pair<char, char>> pairs;
	size_t i = 0;
	while (i < clean.size()) {
		char a = clean[i];

		if (!encrypt) {
			if (clean.size() % 2 != 0) {
				clean += 'X';
			}

			if (i + 1 < clean.size()) {
				pairs.push_back({a, clean[i + 1]});
			}
			i += 2;
		} else if (i + 1 >= clean.size() || clean[i] == clean[i + 1]) {
			pairs.push_back({a, a == 'X' ? 'Q' : 'X'});
			i++;
		} else {
			pairs.push_back({a, clean[i + 1]});
			i += 2;
		}
	}
	return pairs;
}

// Find row and column of a character in the matrix
pair<int, int> findPos(const string& matrix, char c) {
	int idx = matrix.find(c);
	return {idx / 5, idx % 5};
}

// Playfair Encryption & Decryption
string playFair(const string& text, const string& key, bool encrypt) {
	string matrix = prepareKey(key);
	int shift = encrypt ? 1 : 4;
	string result = "";

	vector<pair<char, char>> pairs = prepareText(text, encrypt);

	for (auto [a, b] : pairs) {
		auto [r1, c1] = findPos(matrix, a);
		auto [r2, c2] = findPos(matrix, b);

		if (r1 == r2) {
			result += matrix[r1 * 5 + (c1 + shift) % 5];
			result += matrix[r2 * 5 + (c2 + shift) % 5];
		} else if (c1 == c2) {
			result += matrix[((r1 + shift) % 5) * 5 + c1];
			result += matrix[((r2 + shift) % 5) * 5 + c2];
		} else {
			result += matrix[r1 * 5 + c2];
			result += matrix[r2 * 5 + c1];
		}
	}
	return result;
}
